"""Freeze every page in pages.json, exactly as the local site renders it, into a multi-page artifact.
Each page: computed DOM after scripts ran (canvases turned into images), all same-origin CSS inlined,
images/fonts copied as files, internal links pointing to the frozen copies, plus a small page switcher."""
import asyncio, re, os, shutil, json, sys
from urllib.parse import urljoin, urlparse
from playwright.async_api import async_playwright

REPO = '/home/claude/testaolivier10-del.github.io'
S = '/tmp/claude-0/-home-claude/827fefa7-09dd-576d-acac-c3319d1aab0f/scratchpad'
OUT = S + '/site'
BASE = 'http://localhost:8765/'
HOST = urlparse(BASE).netloc
PAGES = json.load(open(S + '/pages.json'))
ONLY = sys.argv[1:]  # optional subset of paths to refreeze

def out_name(path):
    if path == 'index.html': return 'index.html'
    p = path[:-5] if path.endswith('.html') else path
    if p.endswith('/index'): p = p[:-6]
    return p.replace('/', '-') + '.html'

NAME = {p: out_name(p) for _, p, _ in PAGES}

def nav_html(cur):
    groups = {}
    for g, p, l in PAGES: groups.setdefault(g, []).append((p, l))
    items = ''.join(
        f'<details{" open" if any(p == cur for p, _ in xs) else ""}><summary>{g}</summary>' +
        ''.join(f'<a href="{NAME[p]}"{" aria-current=page" if p == cur else ""}>{l}</a>' for p, l in xs) + '</details>'
        for g, xs in groups.items())
    return ('<div id="lp-pages"><button type="button" id="lp-pages-btn" aria-expanded="false">All pages</button>'
            f'<nav id="lp-pages-list" hidden aria-label="Every page in this preview">{items}</nav></div>'
            '<style>#lp-pages{position:fixed;left:calc(12px + env(safe-area-inset-left,0px));bottom:calc(12px + env(safe-area-inset-bottom,0px));z-index:2147483000;font:600 13px/1.3 system-ui,sans-serif}'
            '#lp-pages-btn{font:800 13px system-ui,sans-serif;background:#0B1A33;color:#fff;border:0;border-radius:999px;padding:9px 14px;cursor:pointer;box-shadow:0 6px 18px rgba(0,0,0,.25)}'
            '#lp-pages-list{position:absolute;left:0;bottom:46px;width:min(300px,86vw);max-height:70vh;overflow:auto;background:#fff;color:#0B1A33;border:1px solid #DFE5EE;border-radius:14px;padding:8px;box-shadow:0 14px 40px rgba(0,0,0,.25)}'
            '#lp-pages-list summary{font-weight:800;padding:7px 8px;cursor:pointer}#lp-pages-list a{display:block;padding:6px 8px 6px 18px;color:#0B1A33;text-decoration:none;border-radius:8px}'
            '#lp-pages-list a:hover{background:#EEF2F7}#lp-pages-list a[aria-current]{background:#E2F4F1;font-weight:800}</style>'
            '<script>(function(){var b=document.getElementById("lp-pages-btn"),l=document.getElementById("lp-pages-list");'
            'b.addEventListener("click",function(){l.hidden=!l.hidden;b.setAttribute("aria-expanded",String(!l.hidden));});})();</script>')

async def freeze(ctx, group, path, label, assets):
    pg = await ctx.new_page()
    url = BASE + path
    try:
        await pg.goto(url, wait_until='networkidle', timeout=45000)
    except Exception as e:
        print('  load warning', path, str(e)[:80])
    await pg.evaluate("async()=>{for(let y=0;y<document.body.scrollHeight;y+=600){window.scrollTo(0,y);await new Promise(r=>setTimeout(r,40))}window.scrollTo(0,0)}")
    await pg.wait_for_timeout(900)
    # Canvases (WebGL included) are captured as they look on screen, then swapped for images.
    n = await pg.evaluate("()=>{const cs=[...document.querySelectorAll('canvas')];cs.forEach((c,i)=>c.setAttribute('data-fz',i));return cs.length}")
    import base64
    for i in range(n):
        el = await pg.query_selector(f'canvas[data-fz="{i}"]')
        try:
            box = await el.bounding_box()
            if box and box['width'] > 2 and box['height'] > 2:
                await pg.evaluate("(i)=>document.querySelector('canvas[data-fz=\"'+i+'\"]').scrollIntoView({block:'center'})", i)
                await pg.wait_for_timeout(1500)
                box = await el.bounding_box()
                sx, sy = await pg.evaluate("()=>[scrollX,scrollY]")
                cdp = await pg.context.new_cdp_session(pg)
                r = await cdp.send('Page.captureScreenshot', {'format': 'png', 'clip': {'x': box['x'] + sx, 'y': box['y'] + sy, 'width': box['width'], 'height': box['height'], 'scale': 1}})
                await cdp.detach()
                png = base64.b64decode(r['data'])
                await pg.evaluate("([i,src])=>{const c=document.querySelector('canvas[data-fz=\"'+i+'\"]');if(c)c.setAttribute('data-shot',src)}", [i, 'data:image/png;base64,' + base64.b64encode(png).decode()])
        except Exception as e:
            print('  canvas capture failed', path, i, str(e)[:160])
    await pg.evaluate("()=>window.scrollTo(0,0)")
    data = await pg.evaluate("""()=>{
      document.querySelectorAll('canvas').forEach(c=>{try{const im=document.createElement('img');im.src=c.getAttribute('data-shot')||c.toDataURL('image/png');
        im.width=c.clientWidth||c.width;im.height=c.clientHeight||c.height;im.style.cssText=c.style.cssText;im.className=c.className;
        im.style.width=(c.clientWidth||c.width)+'px';im.style.height=(c.clientHeight||c.height)+'px';c.replaceWith(im);}catch(e){}});
      const css=[];
      // Source text, not cssRules: Chrome serializes var() shorthands as empty longhands.
      for(const s of document.styleSheets){ if(s.disabled) continue; const n=s.ownerNode;
        const md=(s.media&&s.media.mediaText)||''; if(s.href) css.push({href:s.href, text:null, media:md}); else if(n) css.push({href:location.href, text:n.textContent, media:md}); }
      const root=document.documentElement.cloneNode(true);
      root.querySelectorAll('script,link,style,noscript,meta,iframe').forEach(n=>n.remove());
      root.querySelectorAll('.lp-launch,.lp-tip').forEach(n=>n.remove());
      const b=root.querySelector('body');
      return {css, body:b?b.outerHTML:'<body></body>', cls:document.documentElement.className,
              theme:document.documentElement.getAttribute('data-theme'), lang:document.documentElement.lang, title:document.title};
    }""")
    await pg.close()

    def fix(u, base):
        raw = u.strip().strip('\'"')
        if not raw or raw.startswith(('data:', '#', 'mailto:', 'tel:', 'javascript:', 'blob:')): return u
        a = urljoin(base, raw)
        pr = urlparse(a)
        if pr.netloc != HOST: return a
        rel = pr.path.lstrip('/')
        frag = ('#' + pr.fragment) if pr.fragment else ''
        if rel.endswith('/') or rel == '': rel += 'index.html'
        if rel.endswith('.html'):
            if rel in NAME: return NAME[rel] + frag
            return 'https://levlprep.com/' + rel.replace('index.html', '') + frag
        if os.path.isfile(os.path.join(REPO, rel)):
            assets.add(rel); return rel + frag
        return 'https://levlprep.com/' + rel + frag

    def load(href):
        pr = urlparse(href)
        if pr.netloc != HOST: return ''
        fp = os.path.join(REPO, pr.path.lstrip('/'))
        return open(fp, encoding='utf-8').read() if os.path.isfile(fp) else ''
    def expand(text, base, depth=0):
        def imp(m):
            target = urljoin(base, (m.group(1) or m.group(2) or '').strip('\'"'))
            return expand(load(target), target, depth + 1) if depth < 6 else ''
        text = re.sub(r'@import\s+(?:url\(([^)]+)\)|["\']([^"\']+)["\'])[^;]*;', imp, text)
        return re.sub(r'url\(([^)]+)\)', lambda m: 'url("' + fix(m.group(1), base) + '")', text)
    css = []
    for s in data['css']:
        txt = s['text'] if s['text'] is not None else load(s['href'])
        ex = expand(txt, s['href'])
        md = s.get('media') or ''
        css.append(f'@media {md}{{\n{ex}\n}}' if md and md != 'all' else ex)
    body = data['body']
    def attr(m):
        name, val = m.group(1), m.group(2)
        if name == 'srcset':
            parts = []
            for x in val.split(','):
                bits = x.strip().split()
                if bits: parts.append(' '.join([fix(bits[0], url)] + bits[1:]))
            return f'{name}="{", ".join(parts)}"'
        return f'{name}="{fix(val, url)}"'
    body = re.sub(r'\b(src|srcset|href|xlink:href|poster)="([^"]*)"', attr, body)
    body = re.sub(r'url\(&quot;([^&]+)&quot;\)', lambda m: 'url(&quot;' + fix(m.group(1), url) + '&quot;)', body)
    body = body.replace(' loading="lazy"', '')
    body = re.sub(r'</body>\s*$', nav_html(path) + TRYJS + '</body>', body)
    title = f'{group} · {label}' if path != 'index.html' else 'LevlPrep Homepage Draft'
    html = (f'<meta charset="utf-8">\n<title>{title}</title>\n<style>\n' + '\n'.join(css) + '\n</style>\n'
            f'<script>document.documentElement.className={json.dumps(data["cls"])};'
            + (f'document.documentElement.setAttribute("data-theme",{json.dumps(data["theme"])});' if data['theme'] else '')
            + '</script>\n' + body)
    if path != 'index.html':
        html = '<!doctype html>\n<meta name="viewport" content="width=device-width, initial-scale=1, viewport-fit=cover">\n' + html
    open(os.path.join(OUT, NAME[path]), 'w').write(html)
    return len(html)

async def main():
    os.makedirs(OUT, exist_ok=True)
    man = os.path.join(OUT, 'assets.json')
    assets = set(json.load(open(man))) if ONLY and os.path.exists(man) else set()
    async with async_playwright() as p:
        b = await p.chromium.launch(executable_path='/usr/bin/google-chrome', args=['--no-sandbox'])
        ctx = await b.new_context(viewport={'width': 1440, 'height': 900})
        await ctx.add_init_script("try{localStorage.setItem('levlprep_ai_met','1')}catch(e){}")
        for g, path, label in PAGES:
            if ONLY and path not in ONLY: continue
            n = await freeze(ctx, g, path, label, assets)
            print(f'{NAME[path]:55s} {n//1024:5d} KB')
        await b.close()
    for a in sorted(assets):
        d = os.path.join(OUT, a); os.makedirs(os.path.dirname(d), exist_ok=True); shutil.copy(os.path.join(REPO, a), d)
    json.dump(sorted(assets), open(man, 'w'))
    tot = sum(os.path.getsize(os.path.join(OUT, a)) for a in assets)
    print(len(assets), 'assets', round(tot / 1e6, 1), 'MB')

asyncio.run(main())
